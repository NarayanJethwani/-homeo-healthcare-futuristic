                      disabled={isIngestionRunning}
                      className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl text-xs font-bold uppercase tracking-widest transition-colors cursor-pointer disabled:opacity-50"
                    >
                      {isIngestionRunning ? "Running Parser..." : "Run Ingestion Pipeline"}
                    </button>
                  </div>

                  {/* Right Logs / Summary (8 cols) */}
                  <div className="lg:col-span-8 bg-slate-950/80 border border-slate-900 rounded-[32px] p-6 shadow-2xl space-y-4">
                    <div className="border-b border-slate-900 pb-3 mb-2 flex justify-between items-center">
                      <span className="text-[8px] font-extrabold text-slate-500 uppercase tracking-widest">Pipeline logs & Ingestion Output</span>
                      <span className="text-[8.5px] font-bold text-teal-400 bg-teal-950/40 border border-teal-900/40 px-2 py-0.5 rounded-md">Live Stream</span>
                    </div>

                    {/* Console Logger */}
                    <div className="bg-black/80 border border-slate-900 p-4 rounded-2xl h-56 overflow-y-auto font-mono text-[9px] space-y-1.5 shadow-inner" data-lenis-prevent>
                      {ingestionLogs.length === 0 && (
                        <div className="text-slate-600 italic">Console idle. Awaiting user pipeline trigger...</div>
                      )}
                      {ingestionLogs.map((log: any, i: number) => {
                        let levelStyle = "text-slate-400";
                        if (log.level === "success") levelStyle = "text-emerald-400 font-bold";
                        if (log.level === "warning") levelStyle = "text-amber-400 font-bold";
                        if (log.level === "error") levelStyle = "text-rose-500 font-bold";
                        
                        return (
                          <div key={i} className="leading-relaxed">
                            <span className="text-slate-500 mr-2">[{log.timestamp}]</span>
                            <span className="text-slate-500 mr-2">[{log.stage}]</span>
                            <span className={levelStyle}>{log.message}</span>
                          </div>
                        );
                      })}
                    </div>

                    {/* Summary Result Metrics */}
                    {ingestionSummary && (
                      <div className="bg-slate-900/40 p-4 border border-slate-900 rounded-2xl space-y-3 animate-fadeIn">
                        <div className="flex justify-between items-center border-b border-slate-950 pb-1.5">
                          <span className="text-[10px] font-bold text-emerald-400">Ingestion Summary Report</span>
                          <span className="text-[8.5px] text-slate-500 font-semibold">Parser: V2.1 OCR</span>
                        </div>
                        <div className="grid grid-cols-2 md:grid-cols-4 gap-4 text-xs font-semibold">
                          <div>
                            <span className="text-[8px] text-slate-500 uppercase tracking-wider block">Raw File Size</span>
                            <span>{ingestionSummary.rawFileSizeKb} KB</span>
                          </div>
                          <div>
                            <span className="text-[8px] text-slate-500 uppercase tracking-wider block">Monographs Detected</span>
                            <span>{ingestionSummary.monographsDetected}</span>
                          </div>
                          <div>
                            <span className="text-[8px] text-slate-500 uppercase tracking-wider block">Symptom Fragments</span>
                            <span>{ingestionSummary.symptomsExtractedCount}</span>
                          </div>
                          <div>
                            <span className="text-[8px] text-slate-500 uppercase tracking-wider block">Organ Mappings</span>
                            <span>{ingestionSummary.organAffinitiesMappedCount}</span>
                          </div>
                        </div>

                        {/* Miasm keyword density breakdown */}
                        <div className="space-y-1 pt-1 border-t border-slate-950">
                          <span className="text-[8.5px] text-slate-500 font-extrabold uppercase tracking-widest block">Miasmatic keyword counts</span>
                          <div className="grid grid-cols-5 gap-2 text-center text-[9.5px] font-bold">
                            <div className="bg-slate-950 p-1.5 border border-slate-900 rounded-lg">
                              <div className="text-[7.5px] uppercase tracking-wider text-slate-500 font-extrabold">Psora</div>
                              <div className="text-emerald-400">{ingestionSummary.miasmaticKeywordsCount.psora}</div>
                            </div>
                            <div className="bg-slate-950 p-1.5 border border-slate-900 rounded-lg">
                              <div className="text-[7.5px] uppercase tracking-wider text-slate-500 font-extrabold">Sycosis</div>
                              <div className="text-teal-400">{ingestionSummary.miasmaticKeywordsCount.sycosis}</div>
                            </div>
                            <div className="bg-slate-950 p-1.5 border border-slate-900 rounded-lg">
                              <div className="text-[7.5px] uppercase tracking-wider text-slate-500 font-extrabold">Syphilis</div>
                              <div className="text-rose-400">{ingestionSummary.miasmaticKeywordsCount.syphilis}</div>
                            </div>
                            <div className="bg-slate-950 p-1.5 border border-slate-900 rounded-lg">
                              <div className="text-[7.5px] uppercase tracking-wider text-slate-500 font-extrabold">Tubercular</div>
                              <div className="text-violet-400">{ingestionSummary.miasmaticKeywordsCount.tubercular}</div>
                            </div>
                            <div className="bg-slate-950 p-1.5 border border-slate-900 rounded-lg">
                              <div className="text-[7.5px] uppercase tracking-wider text-slate-500 font-extrabold">Cancer</div>
                              <div className="text-amber-400">{ingestionSummary.miasmaticKeywordsCount.cancerinic}</div>
                            </div>
                          </div>
                        </div>

                        {/* Extracted snippet showcase */}
                        <div className="space-y-2 pt-2 border-t border-slate-950">
                          <span className="text-[8.5px] text-slate-500 font-extrabold uppercase tracking-widest block">Sample Extracted Snippets (Confidence Analysis)</span>
                          <div className="space-y-2">
                            {ingestionSummary.snippets.map((snip: any, idx: number) => (
                              <div key={idx} className="p-3 bg-slate-950 border border-slate-900 rounded-xl text-[10px] space-y-1 select-text">
                                <p className="italic text-slate-400 leading-normal font-serif">&ldquo;{snip.sourceText}&rdquo;</p>
                                <div className="flex justify-between items-center text-[8.5px] font-bold">
                                  <span className="text-slate-500">Extracted: <strong className="text-emerald-400 font-medium">{snip.extractedSymptom}</strong> &rarr; Rubric: <strong className="text-teal-400 font-medium">{snip.matchedRubric}</strong></span>
                                  <span className="text-emerald-400 bg-emerald-950/40 border border-emerald-900/50 px-1.5 py-0.5 rounded">Conf: {snip.confidenceScore}%</span>
                                </div>
                              </div>
                            ))}
                          </div>
                        </div>

                      </div>
                    )}
                  </div>

                </div>
              )}

            </div>
          )}
